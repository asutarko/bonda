For developers

Copy the tokens

Drop these CSS variables into any Bonda build to inherit the colours and fonts.

Colour & font tokens
:root{
  /* core */
  --canvas:#F4F1EB;
  --card:#FFFFFF;
  --ink:#23201C;
  --accent:#3E6E6A;
  --accent-press:#345F5B;
  /* text tints */
  --ink-70:rgba(35,32,28,.70);
  --ink-55:rgba(35,32,28,.55);
  --ink-40:rgba(35,32,28,.40);
  /* lines */
  --line:rgba(35,32,28,.12);
  --line-strong:rgba(35,32,28,.20);
  --focus:rgba(62,110,106,.16);
  /* muted tag tones (bg / text) */
  --tag-teal-bg:#E6EDEC;   --tag-teal-tx:#2E5A56;
  --tag-blue-bg:#E4ECF3;   --tag-blue-tx:#3A5A78;
  --tag-rose-bg:#F3E7EA;   --tag-rose-tx:#7A4651;
  --tag-violet-bg:#EDE8F3; --tag-violet-tx:#574B78;
  --tag-indigo-bg:#E8EAF3; --tag-indigo-tx:#444B78;
  --tag-slate-bg:#E9EAEC;  --tag-slate-tx:#474C52;
  /* type */
  --font-title:"Fraunces",Georgia,serif;
  --font-body:"Plus Jakarta Sans",system-ui,sans-serif;
}
